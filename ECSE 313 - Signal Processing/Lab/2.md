    set(f,'defaultAxesTickLabelInterpreter','none')
    set(f,'defaultLegendInterpreter','none')

    f.Position(3) = 1000;
    f.Position(4) = 800;

`reduce.m`
```matlab
% Define a simple reducer function for use in signals later as it is handy
% https://learn-functional-programming.com/chapters/4.7-sequence-transformations/
function result = reduce(data, reduceF, init)
    result = zeros(1, length(data) + 1);
    result(1) = init;
    for i = 1:length(data)
        result(i+1) = reduceF(result(i), data(i));
    end
end
```

differ = @(s, T) (s(1:numel(s)) - [0, s(1:numel(s)-1)])/T;

% Define our constants, inputs and outputs
systems = {@system1, @system2, @system3};
xlims = [-10, 100];
domain = -1000:1000;
offsets = -200:5:200;
scales = 1:0.1:10;

f = figure;
ff = helpers(f);
t = tiledlayout(3, 2);

% For each system
for system = 1:numel(systems)
    nexttile(t);
    hold on
    % Title for top row
    if system == 1
        title('Varying impulses (normalized)')
    end
    % Plot dirac offsets
    for offset = 1:numel(offsets)
        plot(domain+offsets(offset), outputs_offset{system}{offset});
        % If left column, add row label
        if offset == 1
            ylabel(sprintf('System %d', system))
        end
        xlim(xlims)
    end
    nexttile(t);
    hold on
    % Title for top row
    if system == 1
        title('Varying Scaled Inputs (normalized)')
    end
    % Plot scale offsets
